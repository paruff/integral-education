import useProgress from '@site/src/hooks/useProgress';
import Link from '@docusaurus/Link';
import { useState } from 'react';
import MixedReview from '@site/src/components/MixedReview';
import { readLearnerData } from '@site/src/utils/learnerData';

export function ProgressDashboard() {
  const { data, isAvailable, clear, completedCount, getDaysSinceLastActivity, hasLineProfile, getDueReviews, getMixedReview, recordRetrievalReview } = useProgress();
  const [mixedSession, setMixedSession] = useState(null);
  const [storedData, setStoredData] = useState(null);

  if (!isAvailable) {
    return (
      <div className="alert alert--warning" role="alert">
        <strong>Progress tracking unavailable.</strong> Your browser is in private browsing mode or does not support local storage. Progress will not be saved.
      </div>
    );
  }

  const daysSince = getDaysSinceLastActivity();
  const moduleIds = Object.keys(data.modules || {});
  const totalModules = moduleIds.length;
  const dueReviews = getDueReviews();
  const mixedCards = getMixedReview();
  const mixedModuleCount = new Set(mixedCards.map((c) => c.moduleId)).size;

  return (
    <div>
      {dueReviews.length > 0 && (
        <div className="margin-bottom--md" style={{ padding: '1rem', border: '1px solid var(--ifm-color-warning-dark)', borderRadius: '0.75rem', background: 'var(--ifm-color-warning-lightest)' }}>
          <h3 style={{ marginTop: 0 }}>Due for review</h3>
          <p style={{ fontSize: '0.9rem' }}>Spaced review works best when it's timely — these modules are ready for another pass.</p>
          <ul style={{ marginBottom: 0 }}>
            {dueReviews.map((r) => (
              <li key={r.moduleId}>
                <Link to={`/docs/modules/${r.moduleId}`}>{r.moduleName}</Link>
              </li>
            ))}
          </ul>
        </div>
      )}

      {(mixedSession || mixedCards.length > 0) && (
        <div className="margin-bottom--md" style={{ padding: '1rem', border: '1px solid var(--ifm-color-emphasis-300)', borderRadius: '0.75rem' }}>
          <h3 style={{ marginTop: 0 }}>Mixed review</h3>
          {mixedSession ? (
            <MixedReview
              cards={mixedSession}
              onComplete={(moduleIds) =>
                moduleIds.forEach((id) => recordRetrievalReview(id, data.retrievalSchedule[id]?.moduleName))
              }
            />
          ) : (
            <>
              <p style={{ fontSize: '0.9rem' }}>
                {mixedCards.length} questions from {mixedModuleCount} module{mixedModuleCount === 1 ? '' : 's'}, mixed together. Switching between topics feels harder than reviewing one module at a time, and that effort is what makes recall last.
              </p>
              <button className="button button--primary" onClick={() => setMixedSession(mixedCards)}>
                Start mixed review
              </button>
            </>
          )}
        </div>
      )}

      <div className="margin-top--md margin-bottom--md">
        <p><strong>Modules completed:</strong> {completedCount} of {totalModules}</p>
        {daysSince !== null && (
          <p><strong>Last activity:</strong> {daysSince === 0 ? 'Today' : `${daysSince} day${daysSince === 1 ? '' : 's'} ago`}</p>
        )}

        {hasLineProfile && data.lineProfile && (
          <div className="margin-top--md margin-bottom--md" style={{ padding: '1rem', border: '1px solid var(--ifm-color-emphasis-300)', borderRadius: '0.75rem' }}>
            <h3 style={{ marginTop: 0 }}>Line Profile</h3>
            {['cognitive', 'emotional', 'moral', 'interpersonal', 'somatic', 'spiritual'].map((line) => {
              const level = data.lineProfile[line];
              return (
                <div key={line} style={{ marginBottom: '0.5rem' }}>
                  <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: '0.85rem' }}>
                    <span style={{ fontWeight: 600, textTransform: 'capitalize' }}>{line}</span>
                    <span style={{ color: 'var(--ifm-color-emphasis-600)' }}>{['', 'Not yet', 'Beginning', 'Competent', 'Strong'][level]}</span>
                  </div>
                  <div style={{ height: '6px', background: 'var(--ifm-color-emphasis-200)', borderRadius: '3px' }}>
                    <div style={{ height: '100%', width: `${(level / 4) * 100}%`, background: 'var(--ifm-color-primary)', borderRadius: '3px' }} />
                  </div>
                </div>
              );
            })}
            <Link to="/line-profile" className="margin-top--sm" style={{ fontSize: '0.9rem' }}>Retake Line Profile →</Link>
          </div>
        )}

        {!hasLineProfile && (
          <p className="margin-top--md">
            <Link to="/line-profile">Take your Line Profile →</Link> to see your development across six key lines.
          </p>
        )}
      </div>

      {totalModules === 0 && (
        <div className="alert alert--info" role="alert">
          You haven't started any modules yet. Visit a module page to begin tracking your progress.
        </div>
      )}

      {totalModules > 0 && (
        <table className="margin-top--md">
          <thead>
            <tr>
              <th>Module</th>
              <th>Level</th>
              <th>Completed</th>
              <th>Last Visited</th>
            </tr>
          </thead>
          <tbody>
            {moduleIds.map((id) => {
              const mod = data.modules[id];
              if (!mod) return null;
              return (
                <tr key={id}>
                  <td>{id.replace(/-/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase())}</td>
                  <td>{mod.level ? `Level ${mod.level}` : '—'}</td>
                  <td>{mod.completed ? '✓' : '—'}</td>
                  <td>{mod.lastVisited ? new Date(mod.lastVisited).toLocaleDateString() : '—'}</td>
                </tr>
              );
            })}
          </tbody>
        </table>
      )}

      <div className="margin-top--lg">
        <h3>Your stored data</h3>
        <p style={{ fontSize: '0.9rem' }}>
          Everything below lives only in this browser: module progress, your review schedule and your reflections. Nothing is sent to a server.
        </p>
        <details onToggle={(e) => e.currentTarget.open && setStoredData(readLearnerData(localStorage))}>
          <summary>View my stored data</summary>
          <pre style={{ maxHeight: '20rem', overflow: 'auto', fontSize: '0.8rem' }}>
            {JSON.stringify(storedData ?? {}, null, 2)}
          </pre>
        </details>
        <button
          className="button button--danger margin-top--md"
          onClick={() => {
            if (window.confirm('This deletes your progress, review schedule and reflections from this browser. It cannot be undone. Continue?')) {
              clear();
              setStoredData(readLearnerData(localStorage));
              setMixedSession(null);
            }
          }}
        >
          Clear all my learning data
        </button>
      </div>
    </div>
  );
}

## My Progress

Track your learning journey across all modules. Your progress, review schedule and reflections are stored only in your browser. You can view or clear them at the bottom of this page.
